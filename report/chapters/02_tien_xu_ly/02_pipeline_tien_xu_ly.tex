

\subsection{Tiền xử lý dữ liệu}\label{subsec:preprocessing}
\subsubsection*{1. Làm sạch dữ liệu}
\begin{itemize}
    \item \textbf{Loại bỏ bản ghi trùng lặp:} Tập dữ liệu gốc chứa các giao dịch giống nhau hoàn toàn trên tất cả các đặc trưng. Nhóm quyết định loại bỏ các bản sao thừa và chỉ giữ lại giao dịch đầu tiên. Sau bước này, tập dữ liệu còn lại $283.726$ giao dịch.
    \item \textbf{Xử lý khuyết thiếu và nhiễu:} Ở bước EDA, không phát hiện các giá trị khuyết thiếu (\texttt{Null}) hay giá trị âm bất hợp lệ; đồng thời các điểm ngoại lai trên các thành phần chính $V1$--$V28$ được giữ nguyên vì không biết được ý nghĩa nghiệp vụ của chúng và có thể chúng mang tín hiệu nhận diện hành vi bất thường.
\end{itemize}

\subsubsection*{2. Tạo và biến đổi đặc trưng}
Hai đặc trưng nghiệp vụ gốc là \textbf{Time} và \textbf{Amount} được biến đổi như sau:

\begin{itemize}
    \item \textbf{Biến đổi logarit cho số tiền giao dịch (\texttt{Log\_Amount}):} Do đặc trưng \textbf{Amount} có phân phối lệch phải nghiêm trọng và chứa nhiều giao dịch có giá trị bằng $0$, phép biến đổi $ln(1+x)$ được áp dụng:
    \begin{equation}
        \text{Log\_Amount} = \ln(1 + \text{Amount})
    \end{equation}
    Phép biến đổi này vừa giúp thu hẹp khoảng giá trị ngoại lai, đưa phân phối tiến gần về dạng chuẩn, vừa đảm bảo tính xác định tại các giao dịch có $\text{Amount} = 0$.

    \item \textbf{Biến đổi tuần hoàn cho biến Time (\texttt{Hour\_sin}, \texttt{Hour\_cos}):} Từ biến \textbf{Time} (giây), khung giờ trong ngày được trích xuất theo công thức $\text{Hour} = \lfloor \text{Time} / 3600 \rfloor \bmod 24$. Tuy nhiên, nếu giữ nguyên dạng số tuyến tính ($0$ đến $23$) thì $23\text{h}$ và $0\text{h}$ cách xa nhau $23$ đơn vị trong khi thực tế chúng liền kề nhau. Do đó, đặc trưng \textbf{Hour} được biến đổi lượng giác để có thể tái hiện tính chu kỳ của thời gian:
    \begin{equation}
        \text{Hour\_sin} = \sin\left(\frac{2\pi \times \text{Hour}}{24}\right), \qquad \text{Hour\_cos} = \cos\left(\frac{2\pi \times \text{Hour}}{24}\right)
    \end{equation}

    \item \textbf{Loại bỏ đặc trưng dư thừa:} Sau khi trích xuất thành công các đặc trưng mới, các cột gốc gồm \textbf{Time}, \textbf{Hour} và \textbf{Amount} được loại bỏ khỏi tập dữ liệu để tránh hiện tượng đa cộng tuyến và dư thừa thông tin. Không gian đặc trưng đầu vào cuối cùng bao gồm $30$ thuộc tính ($V1$--$V28$, \texttt{Log\_Amount}, \texttt{Hour\_sin}, \texttt{Hour\_cos} trừ biến mục tiêu \textbf{Class}).
\end{itemize}

\subsubsection*{3. Phân chia dữ liệu}
Với đặc thù dữ liệu mất cân bằng nghiêm trọng (tỷ lệ gian lận dưới $0,2\%$), quy trình thiết lập cơ chế phân chia và chuẩn hóa chặt chẽ:
\begin{itemize}
    \item \textbf{Phân chia tập huấn luyện:} Tập dữ liệu sau khi tiền xử lý được chia thành hai phần độc lập train (80\%) và test (20\%). Kỹ thuật phân tầng đảm bảo tỷ lệ giữa giao dịch gian lận và hợp lệ ở cả hai tập hoàn toàn đồng nhất với tập gốc.
    \item \textbf{ K-Fold Cross-Validation được áp dụng trên tập train:} Kỹ thuật K-Fold được áp dụng trên tập train ($Fold = 5$) để đánh giá tính ổn định của mô hình trước khi test trên tập test. 
\end{itemize}

Nhóm sử dụng kỹ thuật chuẩn hóa dữ liệu \textbf{RobustScaler} thay cho \textbf{StandardScaler} vì StandardScaler được tính bằng trung bình và độ lệch chuẩn, 2 giá trị này vốn nhạy cảm với các outlier mạnh và tập dữ liệu chứa nhiều outlier như đã phân tích ở phần trước

\begin{equation}
    x_{\text{scaled}} = \frac{x - \mu}{\sigma}
\end{equation}
Trong đó:
\begin{itemize}
    \item $x$ là giá trị gốc của đặc trưng
    \item $\mu$ là giá trị trung bình 
    \item $\sigma$ là độ lệch chuẩn được tính trên tập huấn luyện.
\end{itemize}

, trong khi đó RobustScaler sử dụng trung vị và tứ phân vị IQR vốn bền vững trước outlier vì sử dụng trung vị thay vì trung bình để tính toán.
\begin{equation}
    x_{\text{scaled}} = \frac{x - Q_2}{Q_3 - Q_1} = \frac{x - \text{Median}(X)}{\text{IQR}(X)}
\end{equation}
Trong đó:
\begin{itemize}
    \item $Q_1, Q_2, Q_3$ lần lượt là các mức tứ phân vị thứ nhất ($25\%$), thứ hai ($50\%$ -- trung vị) và thứ ba ($75\%$);
    \item $\text{IQR}(X) = Q_3 - Q_1$ là khoảng tứ phân vị (Interquartile Range) của đặc trưng trên tập huấn luyện.
\end{itemize}
